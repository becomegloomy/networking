# Установка утилиты (Windows, через Chocolatey)
choco install nmap

# Определение адреса сети и IP-адреса компьютера
# Вывод: сеть 192.168.0.0/24, IP компьютера 192.168.0.7
ipconfig

# Сканирование сети "вслепую" (без пинга, перебор IP)
nmap -sL 192.168.0.0/24

# Сканирование активных хостов с помощью ICMP (Ping Scan)
nmap -sn 192.168.0.0/24

# Сканирование открытых портов на конкретном хосте (без предварительного пинга)
nmap -r 192.168.0.7

# Попытка определить операционную систему хоста
nmap -O 192.168.0.7
